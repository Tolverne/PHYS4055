
{\Large\bfseries\color{accent}2. Hyperfine, Zeeman and field regimes}
\hrule\smallskip
\begin{multicols}{2}
\subsection*{Magnetic-dipole hyperfine}
\[
H_\mathrm{hfs}=\frac A{\hbar^2}\,\mathbf I\cdot\mathbf J
=\frac A{2\hbar^2}(F^2-I^2-J^2).
\]
\[
\boxed{E_F=\frac A2[F(F+1)-I(I+1)-J(J+1)].}
\]
\(F=|I-J|,\ldots,I+J;\quad 2F+1\) states.
\subsection*{Example \(I=5/2,\ J=1/2\)}
\[
F=2,3,\quad E_2=-7A/4,\quad E_3=5A/4
\]
\[
E_3-E_2=3A,\qquad (2F+1)=5,7.
\]
\subsection*{Zeeman regimes}
Weak field: coupled \(\ket{Fm_F}\) basis,
\[
\Delta E \approx \mu_B g_Fm_FB,
\quad
g_F \approx g_J\frac{F(F+1)+J(J+1)-I(I+1)}
{2F(F+1)}
\]
for \(F>0\), omitting nuclear contribution.
Strong hyperfine-decoupling field:
use approximately \(\ket{Jm_J,Im_I}\); leading
electron shift \(\mu_B g_J m_JB\).
\begin{methodbox}
\textbf{Question workflow:} identify coupling regime,
choose good quantum numbers, write Hamiltonian, evaluate
eigenvalues, check degeneracies and limits.
\end{methodbox}
\end{multicols}
