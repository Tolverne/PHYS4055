
{\Large\bfseries\color{accent}1. Angular momentum, coupling and term symbols}
\hrule\smallskip
\begin{multicols}{2}
\subsection*{Eigenvalues and allowed projections}
\[
J^2\ket{jm}=\hbar^2j(j+1)\ket{jm},\quad
J_z\ket{jm}=\hbar m\ket{jm}.
\]
\(m=-j,-j+1,\ldots,j\); degeneracy \(2j+1\).
\subsection*{Add \(j_1+j_2\)}
\[
j=|j_1-j_2|,|j_1-j_2|+1,\ldots,j_1+j_2.
\]
\[
\sum_j(2j+1)=(2j_1+1)(2j_2+1).
\]
\textbf{Method:} list all \(j\) using triangle rule;
then list \(m\), count/check dimensions.
\subsection*{Coupled and uncoupled bases}
\[
\ket{JM}=\sum_{m_1+m_2=M}
\langle j_1m_1,j_2m_2|JM\rangle
\ket{j_1m_1}\ket{j_2m_2}
\]
(with \(j_1,j_2\) labels fixed).
Use tabulated CG coefficients when available.
\subsection*{Atomic angular momenta}
\(\mathbf j=\mathbf l+\mathbf s\);
\(\mathbf J=\mathbf L+\mathbf S\);
\(\mathbf F=\mathbf I+\mathbf J\).
\[
{}^{2S+1}L_J,\qquad L=0,1,2,3,4\longleftrightarrow S,P,D,F,G.
\]
Do not confuse nuclear \(I\), electronic \(J\), hyperfine \(F\).
\begin{methodbox}
\textbf{When asked for terms:} combine individual
\(l_i,s_i\); respect antisymmetry for equivalent electrons;
list \(L,S,J\); count all microstates.
\end{methodbox}
\end{multicols}
