
{\Large\bfseries\color{accent}5. Atomic structure, radial integrals, lifetimes}
\hrule\smallskip
\begin{multicols}{2}
\subsection*{Equivalent \(d^2\) terms}
\[
{}^3F_{2,3,4},\ {}^3P_{0,1,2},\
{}^1G_4,\ {}^1D_2,\ {}^1S_0.
\]
Microstate check:
\[
21+9+9+5+1=45=\binom{10}{2}.
\]
Hund: maximise \(S\), then \(L\);
for less-than-half-full shells, lowest \(J\)
lies lowest within a term. Approximate ground: \({}^3F_2\).
\subsection*{Radial conventions}
\[
\psi_{nlm}=R_{nl}(r)Y_{lm},\qquad
u_{nl}(r)=rR_{nl}(r).
\]
\[
\int|R|^2r^2\,\diff r=1,\quad \int|u|^2\,\diff r=1.
\]
E1 radial integral:
\[
\mathcal R_{fi}=\int_0^\infty R_f^*(r)\,r\,R_i(r)\,r^2\,\diff r
=\int_0^\infty u_f^*(r)\,r\,u_i(r)\,\diff r.
\]
\subsection*{Perturbing the potential}
\[
\Delta E^{(1)}=\langle\psi|\Delta V|\psi\rangle.
\]
For central \(\Delta V(r)\), integrate
\(|R|^2\Delta V\,r^2\) over \(r\).
\subsection*{Radiative decay}
\[
A_{if}=\frac{\omega_{if}^3|\langle f|\mathbf d|i\rangle|^2}
{3\pi\epsilon_0\hbar c^3},
\qquad
\tau_i=\frac1{\sum_{f}A_{if}}.
\]
Here \(\mathbf d=-e\mathbf r\) and sum is over all
allowed lower states; follow stated averaging conventions.
\[
t_{\mathrm{au}}\simeq2.418884\times10^{-17}\mathrm{s}.
\]
\begin{methodbox}
\textbf{Method:} normalise eigenstates,
compute angular and radial factors,
convert frequencies and dipoles consistently,
sum rates, invert to get lifetime.
\end{methodbox}
\end{multicols}
